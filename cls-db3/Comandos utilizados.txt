2. SEIS PIPELINES DE AGREGACION

// Productos de entrega rápida por categoría
// Pregunta: ¿Cuántos productos con entrega rápida tiene cada categoría, de mayor a menor?

db.catalogo_productos.aggregate([
  { $match: { disponible_entrega_rapida: true } },
  { $group: {
      _id: "$categoria",
      cantidad_productos: { $sum: 1 }
  } },
  { $sort: { cantidad_productos: -1, _id: 1 } }
]).toArray();

// Precio promedio de productos personalizados
// Pregunta: ¿Cuál es el precio base estimado promedio de los productos personalizados por tipo de acero, de mayor a menor?

db.catalogo_productos.aggregate([
  { $match: { es_personalizado: true } },
  { $group: {
      _id: "$tipo_acero",
      precio_promedio: { $avg: "$precio_base_estimado" }
  } },
  { $sort: { precio_promedio: -1, _id: 1 } }
]).toArray();

// Tiempo de fabricación por tipo de acero
// Pregunta: ¿Cuánto tiempo estándar de fabricación suma el catálogo de productos personalizados por tipo de acero?

db.catalogo_productos.aggregate([
  { $match: { es_personalizado: true } },
  { $group: {
      _id: "$tipo_acero",
      tiempo_total_minutos: { $sum: "$tiempo_estandar_minutos" }
  } },
  { $sort: { tiempo_total_minutos: -1, _id: 1 } }
]).toArray();

// Productos con mayor precio base
// Pregunta: ¿Cuáles son los cinco productos con mayor precio base estimado del catálogo?

db.catalogo_productos.aggregate([
  { $sort: { precio_base_estimado: -1, _id: 1 } },
  { $limit: 5 },
  { $project: {
      _id: 1, nombre_producto: 1,
      tipo_acero: 1, precio_base_estimado: 1
  } }
]).toArray();

// Productos de fabricación más corta
// Pregunta: ¿Cuáles son los cinco productos disponibles para entrega rápida con menor tiempo estándar de fabricación?

db.catalogo_productos.aggregate([
  { $match: { disponible_entrega_rapida: true } },
  { $sort: { tiempo_estandar_minutos: 1, _id: 1 } },
  { $limit: 5 },
  { $project: {
      _id: 1, nombre_producto: 1, tiempo_estandar_minutos: 1
  } }
]).toArray();

// Productos personalizados de menor precio
// Pregunta: ¿Cuáles son los cinco productos personalizados de acero AISI 304 con menor precio base estimado?

db.catalogo_productos.aggregate([
  { $match: {
      es_personalizado: true, tipo_acero: "AISI 304"
  } },
  { $sort: { precio_base_estimado: 1, _id: 1 } },
  { $limit: 5 },
  { $project: {
      _id: 1, nombre_producto: 1, precio_base_estimado: 1
  } }
]).toArray();
